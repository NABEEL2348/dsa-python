n=int(input("enter the size of the array"))
arr=[]
for i in range(n):
    arr.append(int(input("Enter the number to append")))
    
print(arr)
pref=1
suff=1
maxi=-999999
for i in range(n):
    if pref==0:
        pref=1
    if suff==0:
        suff=1
    
    pref=pref*arr[i]
    suff=suff*arr[n-i-1]
    maxi=max(maxi,max(pref,suff))
    
    
print(maxi)
