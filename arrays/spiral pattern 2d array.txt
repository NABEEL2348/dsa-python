n=int(input("enter the no of rows to fill"))
m=int(input("enter the no of cols to fill"))
arr=[]
for i in range(n):
    row=[]
    for j in range(m):
        row.append(int(input("enter the number to append")))
    arr.append(row)
    
        
print(arr)
arr1=[]
top=0
bottom=n-1
left=0
right=m-1

while(top<=bottom and left<=right):
    for i in range(left,right+1):
        print(arr[top][i])
        arr1.append(arr[top][i])
    top+=1
    for j in range(top,bottom+1):
        print(arr[j][right])
        arr1.append(arr[j][right])
    right-=1
    for k in range(right, left-1,-1 ):
        print(arr[bottom][k])
        arr1.append(arr[bottom][k])
    bottom-=1
    for l in range(bottom, top-1,-1):
        print(arr[l][left])
        arr1.append(arr[l][left])
    left+=1
print(arr1)
