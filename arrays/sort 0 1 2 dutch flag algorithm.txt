n=int(input("enter the size of the array"))
arr=[]
for i in range(n):
    arr.append(int(input("enter the number to append")))

print(arr)
def swap(low,high,arr):
    temp=arr[low]
    arr[low]=arr[high]
    arr[high]=temp
    
low=0
mid=0
high=n-1
while(mid<=high):
    if(arr[mid]==0):
        swap(low,mid,arr)
        low+=1
        mid+=1
    elif(arr[mid]==1):
        mid+=1
    elif(arr[mid]==2):
        swap(mid,high,arr)
        high-=1
        
        
        
        
        
print(arr)
