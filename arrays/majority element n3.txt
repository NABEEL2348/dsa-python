#include<stdio.h>
int majorityelement(int *arr,int n)
{
    int count=0,count1,candidate1=0,candidate2=0;
    for(int i=0;i<n;i++)
    {
        if(count==0)
        {
            candidate1=arr[i];
            count=1;
        }
        else if(count1==0)
        {
            candidate2=arr[i];
            count1=1;
        }
        else if(arr[i]==candidate1)
        {
            count++;
        }
        else if(arr[i]==candidate2)
        {
            count1++;
        }
        else
        {
            count1--;
            count--;
        }
    }
   
    count=0,count1=0;
    int result[n+1];
    for(int i=0;i<n;i++)
    {
        if(arr[i]==candidate1)
        {
            count++;
        }
        else if(arr[i]==candidate2)
        {
            count1++;
        }
    }
    
    int resultcount=0;
    if(count>n/3)
    {
        result[resultcount++]=candidate1;
    }
    if(count1>n/3)
    {
        result[resultcount++]=candidate2;
    }
  
    for(int i=0;i<resultcount;i++)
    {
        printf("%d ",result[i]);
    }

}
int main()
{
    int arr[]={1,2};
    int n=sizeof(arr)/sizeof(arr[0]);
    majorityelement(arr,n);
}
