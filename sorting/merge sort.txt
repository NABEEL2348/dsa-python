    arr.append(int(input("enter the number to append")))

def merge(arr,low,mid,high):
    
    left=low
    right=mid+1
    temp=[]
    while(left<=mid and right<=high):
        if(arr[left]<=arr[right]):
            temp.append(arr[left])
            left=left+1
        else:
            temp.append(arr[right])
            right=right+1
    
    while(left<=mid):
        temp.append(arr[left])
        left=left+1
        
    while(right<=high):
        temp.append(arr[right])
        right=right+1
    n=len(temp)
    for i in range(n):
        arr[low+i]=temp[i]
   
    return arr
def mergesort(arr,low,high):
    

    if(low>=high):
        return
    mid=(low+high)//2
    mergesort(arr,low,mid)
    mergesort(arr,mid+1,high)
    merge(arr,low,mid,high)
    
    return 0;
mergesort(arr,0,n-1)
