n=int(input("enter the size of the elements"))
arr=[]
for i in range(n):
    arr.append(int(input("enter the number to insert in an array")))
print(arr)

while(n>0):
    i=0
    j=i+1
    while(i<n and j<n):
        if(arr[i]>arr[j]):
            swap=arr[i]
            arr[i]=arr[j]
            arr[j]=swap
        i=i+1
        j=j+1
        
    n=n-1
    
print(arr)
